using System.Data.SqlClient;
using System.Collections.Generic;

namespace StudentCRUD.Models
{
    public class DBHelper
    {
        static string cs = @"Data Source=.;Initial Catalog=TestDB;Integrated Security=True";

        public static List<Student> GetAll()
        {
            var list = new List<Student>();
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlCommand cmd = new SqlCommand("SELECT * FROM Students", con);
                con.Open();
                SqlDataReader dr = cmd.ExecuteReader();
                while (dr.Read())
                    list.Add(new Student { Id = (int)dr["Id"], Name = dr["Name"].ToString(), Age = (int)dr["Age"] });
            }
            return list;
        }

        public static Student GetById(int id)
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlCommand cmd = new SqlCommand("SELECT * FROM Students WHERE Id=@id", con);
                cmd.Parameters.AddWithValue("@id", id);
                con.Open();
                SqlDataReader dr = cmd.ExecuteReader();
                if (dr.Read())
                    return new Student { Id = (int)dr["Id"], Name = dr["Name"].ToString(), Age = (int)dr["Age"] };
            }
            return null;
        }

        public static void Insert(Student s)
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlCommand cmd = new SqlCommand("INSERT INTO Students(Name, Age) VALUES(@n,@a)", con);
                cmd.Parameters.AddWithValue("@n", s.Name);
                cmd.Parameters.AddWithValue("@a", s.Age);
                con.Open();
                cmd.ExecuteNonQuery();
            }
        }

        public static void Update(Student s)
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlCommand cmd = new SqlCommand("UPDATE Students SET Name=@n, Age=@a WHERE Id=@id", con);
                cmd.Parameters.AddWithValue("@n", s.Name);
                cmd.Parameters.AddWithValue("@a", s.Age);
                cmd.Parameters.AddWithValue("@id", s.Id);
                con.Open();
                cmd.ExecuteNonQuery();
            }
        }

        public static void Delete(int id)
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlCommand cmd = new SqlCommand("DELETE FROM Students WHERE Id=@id", con);
                cmd.Parameters.AddWithValue("@id", id);
                con.Open();
                cmd.ExecuteNonQuery();
            }
        }
    }
}
