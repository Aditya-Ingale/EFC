    // To protect from overposting attacks, enable the specific properties you want to bind to.
    // For more details, see http://go.microsoft.com/fwlink/?LinkId=317598.
    [HttpPost]
    [ValidateAntiForgeryToken]
    public async Task<IActionResult> Edit(int? id, [Bind("Id,Name,Gender")] Employee employee)
    {
        if (id != employee.Id)
        {
            return NotFound();
        }

        if (ModelState.IsValid)
        {
            try
            {
                _context.Update(employee);
                await _context.SaveChangesAsync();
            }
            catch (DbUpdateConcurrencyException)
            {
                if (!EmployeeExists(employee.Id))
                {
                    return NotFound();
                }
                else
                {
                    throw;
                }
            }
            return RedirectToAction(nameof(Index));
        }
        return View(employee);
    }

        return _context.Employees.Any(e => e.Id == id);

using System.Data.SqlClient;
using System.Collections.Generic;

namespace StudentCRUD.Models
{
    public class DBHelper
    {
        static string cs = @"Data Source=.;Initial Catalog=TestDB;Integrated Security=True";

        public static List<Student> GetAll()
        {
            var list = new List<Student>();
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlCommand cmd = new SqlCommand("SELECT * FROM Students", con);
                con.Open();
                SqlDataReader dr = cmd.ExecuteReader();
                while (dr.Read())
                    list.Add(new Student { Id = (int)dr["Id"], Name = dr["Name"].ToString(), Age = (int)dr["Age"] });
            }
            return list;
        }

        public static Student GetById(int id)
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlCommand cmd = new SqlCommand("SELECT * FROM Students WHERE Id=@id", con);
                cmd.Parameters.AddWithValue("@id", id);
                con.Open();
                SqlDataReader dr = cmd.ExecuteReader();
                if (dr.Read())
                    return new Student { Id = (int)dr["Id"], Name = dr["Name"].ToString(), Age = (int)dr["Age"] };
            }
            return null;
        }

        public static void Insert(Student s)
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlCommand cmd = new SqlCommand("INSERT INTO Students(Name, Age) VALUES(@n,@a)", con);
                cmd.Parameters.AddWithValue("@n", s.Name);
                cmd.Parameters.AddWithValue("@a", s.Age);
                con.Open();
                cmd.ExecuteNonQuery();
            }
        }

        public static void Update(Student s)
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlCommand cmd = new SqlCommand("UPDATE Students SET Name=@n, Age=@a WHERE Id=@id", con);
                cmd.Parameters.AddWithValue("@n", s.Name);
                cmd.Parameters.AddWithValue("@a", s.Age);
                cmd.Parameters.AddWithValue("@id", s.Id);
                con.Open();
                cmd.ExecuteNonQuery();
            }
        }

        public static void Delete(int id)
        {
            using (SqlConnection con = new SqlConnection(cs))
            {
                SqlCommand cmd = new SqlCommand("DELETE FROM Students WHERE Id=@id", con);
                cmd.Parameters.AddWithValue("@id", id);
                con.Open();
                cmd.ExecuteNonQuery();
            }
        }
    }
}

// Assignment 4 - Question 1: Implementing a News Feed Application
// Demonstrates the use of delegates and events in C#.
//
// - NewsArticle: represents a single news article
// - NewsArticleCollection: represents a collection of news articles,
//   raises events when articles are added or removed, and supports
//   filtering articles by category
// - Program: console app that subscribes to the events and exercises
//   the collection
//
// To run (requires .NET SDK):
//   dotnet run --project Task1_NewsFeedApp.csproj
// ==========================================================================

using System;
using System.Collections.Generic;
using System.Linq;

namespace NewsFeedApp
{
    /// <summary>
    /// Represents a single news article.
    /// </summary>
    public class NewsArticle
    {
        public int Id { get; set; }
        public string Title { get; set; }
        public string Content { get; set; }
        public string Category { get; set; }
        public DateTime PublishedDate { get; set; }

        public NewsArticle(int id, string title, string content, string category, DateTime publishedDate)
        {
            Id = id;
            Title = title;
            Content = content;
            Category = category;
            PublishedDate = publishedDate;
        }

        public override string ToString()
        {
            return $"[{Id}] ({Category}) \"{Title}\" - published {PublishedDate:d}";
        }
    }

    // Delegate that defines the signature for article add/remove event handlers.
    public delegate void ArticleChangedEventHandler(object sender, ArticleChangedEventArgs e);

    /// <summary>
    /// Custom event args carrying information about the article that changed
    /// and the type of change that occurred.
    /// </summary>
    public class ArticleChangedEventArgs : EventArgs
    {
        public NewsArticle Article { get; }
        public string ChangeType { get; } // "Added" or "Removed"

        public ArticleChangedEventArgs(NewsArticle article, string changeType)
        {
            Article = article;
            ChangeType = changeType;
        }
    }

    /// <summary>
    /// Represents a collection of news articles. Raises events whenever
    /// an article is added to or removed from the collection, and allows
    /// subscribers (e.g. the console app) to react - for example, by
    /// notifying the user.
    /// </summary>
    public class NewsArticleCollection
    {
        private readonly List<NewsArticle> _articles = new List<NewsArticle>();

        // Events that other parts of the application can subscribe to.
        public event ArticleChangedEventHandler ArticleAdded;
        public event ArticleChangedEventHandler ArticleRemoved;

        public int Count => _articles.Count;

        public void AddArticle(NewsArticle article)
        {
            if (article == null) throw new ArgumentNullException(nameof(article));

            _articles.Add(article);

            // Raise the ArticleAdded event, notifying all subscribers.
            OnArticleAdded(new ArticleChangedEventArgs(article, "Added"));
        }

        public bool RemoveArticle(int articleId)
        {
            NewsArticle article = _articles.FirstOrDefault(a => a.Id == articleId);
            if (article == null)
            {
                return false;
            }

            _articles.Remove(article);

            // Raise the ArticleRemoved event, notifying all subscribers.
            OnArticleRemoved(new ArticleChangedEventArgs(article, "Removed"));
            return true;
        }

        /// <summary>
        /// Returns all articles that belong to the given category
        /// (case-insensitive match).
        /// </summary>
        public IEnumerable<NewsArticle> FilterByCategory(string category)
        {
            return _articles
                .Where(a => string.Equals(a.Category, category, StringComparison.OrdinalIgnoreCase))
                .OrderByDescending(a => a.PublishedDate);
        }

        public IEnumerable<NewsArticle> GetAllArticles()
        {
            return _articles.OrderByDescending(a => a.PublishedDate);
        }

        // Protected "on" methods - standard .NET pattern for raising events.
        protected virtual void OnArticleAdded(ArticleChangedEventArgs e)
        {
            ArticleAdded?.Invoke(this, e);
        }

        protected virtual void OnArticleRemoved(ArticleChangedEventArgs e)
        {
            ArticleRemoved?.Invoke(this, e);
        }
    }

    public class Program
    {
        public static void Main(string[] args)
        {
            var feed = new NewsArticleCollection();

            // Subscribe to the events using delegates - the event handlers
            // below simply print a notification to the console, but any
            // subscriber logic (logging, email alerts, UI refresh, etc.)
            // could go here instead.
            feed.ArticleAdded += Feed_ArticleAdded;
            feed.ArticleRemoved += Feed_ArticleRemoved;

            Console.WriteLine("=== Adding articles to the news feed ===\n");

            feed.AddArticle(new NewsArticle(1, "Local Team Wins Championship", "...", "Sports", new DateTime(2024, 3, 1)));
            feed.AddArticle(new NewsArticle(2, "New Smartphone Released", "...", "Technology", new DateTime(2024, 3, 3)));
            feed.AddArticle(new NewsArticle(3, "Stock Market Rallies", "...", "Business", new DateTime(2024, 3, 4)));
            feed.AddArticle(new NewsArticle(4, "City Council Passes New Budget", "...", "Politics", new DateTime(2024, 3, 5)));
            feed.AddArticle(new NewsArticle(5, "AI Breakthrough Announced", "...", "Technology", new DateTime(2024, 3, 6)));

            Console.WriteLine($"\nTotal articles in feed: {feed.Count}");

            Console.WriteLine("\n=== Filtering articles by category: Technology ===\n");
            foreach (var article in feed.FilterByCategory("Technology"))
            {
                Console.WriteLine(article);
            }

            Console.WriteLine("\n=== Removing an article ===\n");
            feed.RemoveArticle(3);

            Console.WriteLine($"\nTotal articles in feed: {feed.Count}");

            Console.WriteLine("\n=== All remaining articles ===\n");
            foreach (var article in feed.GetAllArticles())
            {
                Console.WriteLine(article);
            }

            Console.WriteLine("\nDone. Press any key to exit.");
        }

        // Event handler methods matching the ArticleChangedEventHandler delegate signature.
        private static void Feed_ArticleAdded(object sender, ArticleChangedEventArgs e)
        {
            Console.WriteLine($"[NOTIFICATION] New article added -> {e.Article}");
        }

        private static void Feed_ArticleRemoved(object sender, ArticleChangedEventArgs e)
        {
            Console.WriteLine($"[NOTIFICATION] Article removed -> {e.Article}");
        }
    }
}

                .Where(o => o.ShippedDate.HasValue
                         && o.ShippedDate.Value.Year == 2021
                         && o.ShippedDate.Value.Month == 10)
                .ToList();

            foreach (var o in ordersShippedOctober2021)
                Console.WriteLine($"Order #{o.OrderId} - shipped {o.ShippedDate:d}");

            // ------------------------------------------------------------------
            // e) All orders shipped to customers in the USA
            // ------------------------------------------------------------------
            Console.WriteLine("\ne) Orders shipped to customers in the USA");
            Console.WriteLine("------------------------------------------");
            var ordersToUsa =
                from o in orders
                join c in customers on o.CustomerId equals c.CustomerId
                where c.Country == "USA"
                select new { o.OrderId, Customer = c.FullName, o.ShippedDate };

            foreach (var o in ordersToUsa)
                Console.WriteLine($"Order #{o.OrderId} - {o.Customer} - shipped {(o.ShippedDate.HasValue ? o.ShippedDate.Value.ToShortDateString() : "not shipped yet")}");

            // ------------------------------------------------------------------
            // f) Products ordered at least once, with total quantity and
            //    total revenue generated by each product
            // ------------------------------------------------------------------
            Console.WriteLine("\nf) Products ordered, with total quantity and total revenue");
            Console.WriteLine("------------------------------------------------------------");
            var productSales =
                from od in orderDetails
                group od by od.ProductId into g
                join p in products on g.Key equals p.ProductId
                select new
                {
                    ProductName = p.ProductName,
                    TotalQuantity = g.Sum(x => x.Quantity),
                    TotalRevenue = g.Sum(x => x.Quantity * x.UnitPrice)
                };

            foreach (var ps in productSales.OrderByDescending(x => x.TotalRevenue))
                Console.WriteLine($"{ps.ProductName,-30} Qty: {ps.TotalQuantity,-5} Revenue: {ps.TotalRevenue:C}");

            // ------------------------------------------------------------------
            // g) Top 5 customers by total amount spent, with number of
            //    orders placed and average order amount
            // ------------------------------------------------------------------
            Console.WriteLine("\ng) Top 5 customers by total amount spent");
            Console.WriteLine("------------------------------------------");

            // First compute the total amount of each individual order.
            var orderTotals =
                from o in orders
                join od in orderDetails on o.OrderId equals od.OrderId
                group od by o.OrderId into g
                select new
                {
                    OrderId = g.Key,
                    CustomerId = orders.First(o => o.OrderId == g.Key).CustomerId,
                    OrderTotal = g.Sum(x => x.Quantity * x.UnitPrice)
                };

            var topCustomers =
                from ot in orderTotals
                group ot by ot.CustomerId into g
                join c in customers on g.Key equals c.CustomerId
                select new
                {
                    Customer = c.FullName,
                    TotalSpent = g.Sum(x => x.OrderTotal),
                    OrderCount = g.Count(),
                    AverageOrderAmount = g.Average(x => x.OrderTotal)
                } into result
                orderby result.TotalSpent descending
                select result;

            foreach (var tc in topCustomers.Take(5))
                Console.WriteLine($"{tc.Customer,-15} Total: {tc.TotalSpent,10:C}  Orders: {tc.OrderCount}  Avg/Order: {tc.AverageOrderAmount:C}");

            // ------------------------------------------------------------------
            // h) The most popular product by number of times it has
            //    been ordered (i.e. appears in the most order details)
            // ------------------------------------------------------------------
            Console.WriteLine("\nh) Most popular product (by number of times ordered)");
            Console.WriteLine("------------------------------------------------------");
            var mostPopularProduct =
                (from od in orderDetails
                 group od by od.ProductId into g
                 join p in products on g.Key equals p.ProductId
                 select new
                 {
                     ProductName = p.ProductName,
                     TimesOrdered = g.Count()
                 })
                .OrderByDescending(x => x.TimesOrdered)
                .First();

            Console.WriteLine($"{mostPopularProduct.ProductName} - ordered {mostPopularProduct.TimesOrdered} time(s)");

            Console.WriteLine("\nDone.");
        }
    }
