-- --------------------------------------------------------------
-- Q1. Create minimum 3 tables without any Key constraints (6)
-- --------------------------------------------------------------

CREATE TABLE Category (
    category_id INT,
    category_name VARCHAR(50)
);

CREATE TABLE Medicines (
    medicine_id INT,
    medicine_name VARCHAR(100),
    price DECIMAL(10,2),
    stock_qty INT,
    category_id INT
);

CREATE TABLE Sales (
    sale_id INT,
    medicine_id INT,
    quantity INT,
    sale_date DATE
);

-- --------------------------------------------------------------
-- Q2. Add Primary Key to each table (3)
-- --------------------------------------------------------------

ALTER TABLE Category
    ADD PRIMARY KEY (category_id);

ALTER TABLE Medicines
    ADD PRIMARY KEY (medicine_id);

ALTER TABLE Sales
    ADD PRIMARY KEY (sale_id);

-- --------------------------------------------------------------
-- Q3. Connect at least 2 tables using Foreign Key (2)
-- --------------------------------------------------------------

ALTER TABLE Medicines
    ADD CONSTRAINT fk_medicines_category
    FOREIGN KEY (category_id) REFERENCES Category(category_id);

ALTER TABLE Sales
    ADD CONSTRAINT fk_sales_medicine
    FOREIGN KEY (medicine_id) REFERENCES Medicines(medicine_id);

-- --------------------------------------------------------------
-- Q12. Procedure with one INPUT and one OUTPUT parameter (3)
-- --------------------------------------------------------------

DELIMITER //

CREATE PROCEDURE GetStockByMedicine(
    IN p_medicine_id INT,
    OUT p_stock INT
)
BEGIN
    SELECT stock_qty INTO p_stock
    FROM Medicines
    WHERE medicine_id = p_medicine_id;
END //

DELIMITER ;

-- To call the procedure:
CALL GetStockByMedicine(201, @stock);
SELECT @stock;

-- --------------------------------------------------------------
-- Q13. Trigger of BEFORE UPDATE on any one table (3)
-- --------------------------------------------------------------

DELIMITER //

CREATE TRIGGER before_medicine_update
BEFORE UPDATE ON Medicines
FOR EACH ROW
BEGIN
    IF NEW.stock_qty < 0 THEN
        SET NEW.stock_qty = 0;
    END IF;
END //

DELIMITER ;
